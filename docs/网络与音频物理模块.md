### 1.6 使用示例

```cpp
// HTTP GET请求
auto request = new HttpRequest();
request->setUrl("http://api.example.com/data");
request->setRequestType(HttpRequest::Type::GET);
request->setResponseCallback([](HttpClient* client, HttpResponse* response) {
    if (response->isSucceed()) {
        auto data = response->getResponseData();
        std::string str(data->begin(), data->end());
        CCLOG("Response: %s", str.c_str());
    }
});
HttpClient::getInstance()->send(request);
request->release();

// WebSocket
class MyDelegate : public WebSocket::Delegate {
    virtual void onOpen(WebSocket* ws) { CCLOG("Connected"); }
    virtual void onMessage(WebSocket* ws, const Data& data) { /* 处理消息 */ }
    virtual void onClose(WebSocket* ws) { CCLOG("Closed"); }
    virtual void onError(WebSocket* ws, const ErrorCode& error) { CCLOG("Error"); }
};
```

### 3.6 使用示例

```cpp
// 创建物理场景
auto scene = Scene::createWithPhysics();
scene->getPhysicsWorld()->setGravity(Vect(0, -98.0f));
// 开启调试绘制
scene->getPhysicsWorld()->setDebugDrawMask(PhysicsWorld::DEBUGDRAW_ALL);

// 创建物理刚体
auto body = PhysicsBody::createBox(Size(100, 100));
body->setDynamic(true);
body->setCategoryBitmask(0x01);
body->setCollisionBitmask(0x02);
body->setContactTestBitmask(0xFFFFFFFF);

auto sprite = Sprite::create("box.png");
sprite->setPhysicsBody(body);
scene->addChild(sprite);

// 碰撞监听
auto contactListener = EventListenerPhysicsContact::create();
contactListener->onContactBegin = [](PhysicsContact& contact) {
    CCLOG("Contact begin!");
    return true;
};
_eventDispatcher->addEventListenerWithSceneGraphPriority(contactListener, scene);
```
