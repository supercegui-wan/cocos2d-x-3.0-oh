### 1.2 核心基类

#### Action
```cpp
class Action : public Ref, public Clonable {
public:
    static const int INVALID_TAG = -1;
    
    virtual Action* clone() const = 0;      // 克隆
    virtual Action* reverse() const = 0;    // 创建反向动作
    virtual bool isDone() const;            // 是否完成
    virtual void startWithTarget(Node* target); // 开始执行
    virtual void stop();                    // 停止
    virtual void step(float dt);            // 步进（内部调用update）
    virtual void update(float time);        // 更新（time: 0~1）
    
    Node* getTarget() const;
    void setTarget(Node* target);
    int getTag() const;
    void setTag(int tag);
    
protected:
    Node* _originalTarget;
    Node* _target;
    int _tag;
};
```

#### FiniteTimeAction
```cpp
class FiniteTimeAction : public Action {
    float getDuration() const;
    void setDuration(float duration);
    virtual FiniteTimeAction* reverse() const = 0;
    virtual FiniteTimeAction* clone() const = 0;
protected:
    float _duration;  // 持续时间（秒）
};
```

#### Speed
```cpp
class Speed : public Action {
    static Speed* create(ActionInterval* action, float speed);
    float getSpeed() const;
    void setSpeed(float speed);          // speed>1 加速, speed<1 减速
    void setInnerAction(ActionInterval*);
    ActionInterval* getInnerAction() const;
};
```

#### Follow
```cpp
class Follow : public Action {
    static Follow* create(Node* followedNode, const Rect& rect = Rect::ZERO);
    bool isBoundarySet() const;
    void setBoudarySet(bool value);
};
```

## 2. 瞬时动作 (ActionInstant)

### 2.1 可见性
```cpp
// Show - 显示节点
auto show = Show::create();

// Hide - 隐藏节点
auto hide = Hide::create();

// ToggleVisibility - 切换可见性
auto toggle = ToggleVisibility::create();
```

### 2.2 变换
```cpp
// FlipX - 水平翻转
auto flipX = FlipX::create(true/false);

// FlipY - 垂直翻转
auto flipY = FlipY::create(true/false);

// Place - 放置到指定位置
auto place = Place::create(Point(100, 200));
```

### 2.3 回调
```cpp
// CallFunc - 函数回调
auto callFunc = CallFunc::create(CC_CALLBACK_0(MyClass::onComplete, this));
// 或
auto callFunc = CallFunc::create([&]() { /* lambda */ });

// RemoveSelf - 从父节点移除自身
auto remove = RemoveSelf::create();  // 等价于 create(true)
// create(bool isNeedCleanUp = true)  — isNeedCleanUp 控制是否清理动作
```

## 3. 持续动作 (ActionInterval)

### 3.1 基本变换动作
```cpp
// 移动
MoveTo::create(duration, Point(x, y));
MoveBy::create(duration, Point(dx, dy));

// 旋转
RotateTo::create(duration, angle);
RotateTo::create(duration, deltaAngleX, deltaAngleY);  // 2D 旋转（X/Y 轴分量）
RotateBy::create(duration, deltaAngle);
RotateBy::create(duration, deltaAngleX, deltaAngleY);  // 2D 旋转（X/Y 轴分量）

// 缩放
ScaleTo::create(duration, scale);        // 统一缩放
ScaleTo::create(duration, scaleX, scaleY);
ScaleBy::create(duration, scale);

// 淡入淡出
FadeIn::create(duration);
FadeOut::create(duration);
FadeTo::create(duration, opacity);

// 着色
TintTo::create(duration, GLubyte red, GLubyte green, GLubyte blue);
TintBy::create(duration, GLshort deltaRed, GLshort deltaGreen, GLshort deltaBlue);

// 闪烁
Blink::create(duration, times);
```

### 3.2 组合动作
```cpp
// Sequence - 顺序执行
auto seq = Sequence::create(action1, action2, action3, nullptr);

// Spawn - 并行执行
auto spawn = Spawn::create(action1, action2, nullptr);

// Repeat - 重复N次
auto repeat = Repeat::create(action, times);
auto repeatForever = RepeatForever::create(action);

// DelayTime - 延迟等待
auto delay = DelayTime::create(seconds);
```

### 3.3 高级动作
```cpp
// 贝塞尔曲线移动
BezierTo::create(duration, ccBezierConfig);
BezierBy::create(duration, ccBezierConfig);

// 跳跃
JumpTo::create(duration, Point(x,y), height, jumps);
JumpBy::create(duration, Point(dx,dy), height, jumps);

// 样条曲线
CardinalSplineTo::create(duration, pointArray, tension);
CardinalSplineBy::create(duration, pointArray, tension);
CatmullRomTo::create(duration, pointArray);
CatmullRomBy::create(duration, pointArray);

// 进度条
ProgressTo::create(duration, percent);
ProgressFromTo::create(duration, fromPercent, toPercent);

// 目标动作
TargetedAction::create(target, action);

// 帧动画
Animate::create(animation);
```

## 4. 缓动动作 (ActionEase)

**文件**: `cocos/2d/CCActionEase.h`

缓动动作包装另一个 ActionInterval，修改其时间曲线。

### 4.1 缓动类型

| 类别 | 动作 | 说明 |
|------|------|------|
| 基础 | EaseIn | 缓入 (t^rate) |
| | EaseOut | 缓出 |
| | EaseInOut | 缓入缓出 |
| 指数 | EaseExponentialIn/Out/InOut | 指数缓动 |
| 正弦 | EaseSineIn/Out/InOut | 正弦缓动 |
| 弹性 | EaseElasticIn/Out/InOut | 弹性缓动 |
| 弹跳 | EaseBounceIn/Out/InOut | 弹跳缓动 |
| 回退 | EaseBackIn/Out/InOut | 回退缓动 |
| 贝塞尔 | EaseBezierAction | 自定义贝塞尔 |

### 4.2 使用示例
```cpp
auto move = MoveTo::create(2.0f, Point(300, 200));
auto easeMove = EaseInOut::create(move, 2.0f);
sprite->runAction(easeMove);

// 自定义贝塞尔
auto easeBezier = EaseBezierAction::create(action);
easeBezier->setBezierParamer(0.5f, 0.0f, 0.5f, 1.0f);  // 通过 setBezierParamer 设置贝塞尔参数
```

### 4.3 tweenfunc 命名空间与 TweenType 枚举

**文件**: `cocos/2d/CCTweenFunction.h`

```cpp
namespace tweenfunc {
    // 缓动函数族，共 33 种缓动类型
    // 每个函数接受 float time (0~1) 返回 float (0~1)
    
    enum TweenType {
        CustomEase = -1,
        
        // 线性
        Linear = 0,
        
        // 二次 (Quadratic)
        Sine_EaseIn, Sine_EaseOut, Sine_EaseInOut,
        
        // 三次 (Cubic)
        Quad_EaseIn, Quad_EaseOut, Quad_EaseInOut,
        
        // 四次 (Quartic)
        Cubic_EaseIn, Cubic_EaseOut, Cubic_EaseInOut,
        
        // 五次 (Quintic)
        Quart_EaseIn, Quart_EaseOut, Quart_EaseInOut,
        
        // 圆形 (Circle)
        Quint_EaseIn, Quint_EaseOut, Quint_EaseInOut,
        
        // 指数 (Exponential)
        Expo_EaseIn, Expo_EaseOut, Expo_EaseInOut,
        
        // 弹性 (Elastic)
        Elastic_EaseIn, Elastic_EaseOut, Elastic_EaseInOut,
        
        // 回退 (Back)
        Back_EaseIn, Back_EaseOut, Back_EaseInOut,
        
        // 弹跳 (Bounce)
        Bounce_EaseIn, Bounce_EaseOut, Bounce_EaseInOut,
        
        // 总数
        TweenType_Count = 33,
    };
    
    // 使用方式
    float result = tweenfunc::sineEaseIn(time);
    // 或通过 ActionEase 系列动作间接使用
}
```

## 6. 使用示例

### 6.1 基本动作链
```cpp
auto move = MoveTo::create(2.0f, Point(300, 200));
auto fade = FadeOut::create(1.0f);
auto spawn = Spawn::create(move, fade, nullptr);
auto delay = DelayTime::create(0.5f);
auto remove = RemoveSelf::create();
auto seq = Sequence::create(spawn, delay, remove, nullptr);
sprite->runAction(seq);
```

### 6.2 缓动结合
```cpp
auto move = MoveBy::create(2.0f, Point(200, 0));
auto easeMove = EaseElasticOut::create(move->clone());
auto repeat = RepeatForever::create(easeMove);
sprite->runAction(repeat);
```

### 6.3 速度控制
```cpp
auto move = MoveTo::create(5.0f, Point(500, 0));
auto speed = Speed::create(move, 2.0f);  // 2倍速
sprite->runAction(speed);
```

## 7. CCActionManager - 动作管理器

**文件**: `cocos/2d/CCActionManager.h`

```cpp
class ActionManager : public Ref {
    // 添加/移除动作
    void addAction(Action* action, Node* target, bool paused);
    void removeAllActionsFromTarget(Node* target);
    void removeAction(Action* action);
    void removeActionByTag(int tag, Node* target);
    
    // 查询
    Action* getActionByTag(int tag, const Node* target) const;
    ssize_t getNumberOfRunningActionsInTarget(const Node* target) const;
    
    // 控制
    void pauseTarget(Node* target);
    void resumeTarget(Node* target);
    void pauseAllRunningActions();          // 暂停所有运行中的动作
    void resumeTargets(const Vector<Node*>& targetsToResume);  // 恢复指定目标列表
    void removeAllActions();                // 移除所有目标的所有动作
    void update(float dt);  // 每帧更新所有动作
};
```
